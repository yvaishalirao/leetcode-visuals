#722

Remove Comments

View on LeetCode ↗
A character-by-character state machine: track whether a block comment (/* … */) is currently open. "//" outside a block comment swallows the rest of the line; "/*" opens a block comment (which can span multiple lines, silently deleting the newlines between); everything else, when not inside a block comment, goes into the current output line's buffer.
Solution code
class Solution(object):
    def removeComments(self, source):
        res, buffer, block_comment_open = [], '', False
        for line in source:
            i = 0
            while i < len(line):
                char = line[i]
                if char == '/' and (i + 1) < len(line) and line[i + 1] == '/' and not block_comment_open:
                    i = len(line)
                elif char == '/' and (i + 1) < len(line) and line[i + 1] == '*' and not block_comment_open:
                    block_comment_open = True
                    i += 1
                elif char == '*' and (i + 1) < len(line) and line[i + 1] == '/' and block_comment_open:
                    block_comment_open = False
                    i += 1
                elif not block_comment_open:
                    buffer += char
                i += 1
            if buffer and not block_comment_open:
                res.append(buffer)
                buffer = ''
        return res
