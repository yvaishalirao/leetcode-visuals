Find Mode in Binary Search Tree

View on LeetCode ↗
Given the root of a BST with duplicates, return all the mode(s) — the most frequently occurring value(s). In-order traversal of a BST visits values in sorted order, so every run of equal values is adjacent — no hashmap needed, just track a running streak against prev.
Solution code
class Solution(object):
    prev = None
    max_count = 0
    current_count = 0
    result = []

    def findMode(self, root):
        self.dfs(root)
        return self.result

    def dfs(self, node):
        if not node: return
        self.dfs(node.left)
        self.current_count = 1 if node.val != self.prev else self.current_count + 1
        if self.current_count == self.max_count:
            self.result.append(node.val)
        elif self.current_count > self.max_count:
            self.result = [node.val]
            self.max_count = self.current_count
        self.prev = node.val
        self.dfs(node.right)
